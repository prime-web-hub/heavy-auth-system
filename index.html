async function loginWithMobile() { const emailForOTP = prompt("OTP प्राप्त करने के लिए अपनी ईमेल आईडी डालें:"); if (!emailForOTP) return; const messageDiv = document.getElementById('message'); messageDiv.innerText = "ईमेल पर OTP भेजा जा रहा है..."; messageDiv.className = ""; try { const response = await fetch(`${API_URL}/send-otp`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: emailForOTP }) }); const data = await response.json(); if (response.ok) { const userOTP = prompt("आपकी ईमेल पर 6-अंकों का OTP भेजा गया है। कृपया यहाँ डालें:"); if (!userOTP) return; // अब OTP वेरिफाई करने के लिए बैकएंड को भेजें const verifyRes = await fetch(`${API_URL}/verify-otp`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email: emailForOTP, otp: userOTP }) }); const verifyData = await verifyRes.json(); if (verifyRes.ok) { messageDiv.innerText = "लॉगिन सफल रहा! (Email OTP Verified)"; messageDiv.className = "success"; if(verifyData.token) localStorage.setItem('authToken', verifyData.token); } else { messageDiv.innerText = verifyData.error || "गलत OTP डाला गया है।"; messageDiv.className = "error"; } } else { messageDiv.innerText = data.error || "OTP भेजने में विफलता।"; messageDiv.className = "error"; } } catch (error) { messageDiv.innerText = "सर्वर से कनेक्शन नहीं हो पाया।"; messageDiv.className = "error"; } }
